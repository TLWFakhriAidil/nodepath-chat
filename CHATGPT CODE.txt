     function sendGroupMessage($group_name, $wa_text, $instance) {
        $url = 'https://app.whacenter.com/api/sendGroup'; // Group message endpoint
    
        $ch = curl_init($url);
    
        // Prepare the data for the group message
        $data = array(
            'device_id' => $instance,
            'group' => $group_name, // The name of the group
            'message' => $wa_text   // The message to send
        );
    
        // Set cURL options
        curl_setopt($ch, CURLOPT_POST, true);
        curl_setopt($ch, CURLOPT_POSTFIELDS, $data);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    
        // Execute the request
        $result = curl_exec($ch);
        curl_close($ch);
    
        // Return the result
        return $result;
    }

    public function checkusername(Request $request) {
    $response = [];

    // Check if username is provided and exists
    if ($request->has('username')) {
        $username = $request->get('username');
        $response['username'] = ZUser::where('username', $username)->exists() ? 'Existed' : 'Success';
    }

    // Check if email is provided and exists
    if ($request->has('email')) {
        $email = $request->get('email');
        $response['email'] = ZUser::where('email', $email)->exists() ? 'Existed' : 'Success';
    }

    // Check if phone is provided and exists
    if ($request->has('phone')) {
        $phone = $request->get('phone');
        $response['phone'] = ZUser::where('no_tel', $phone)->exists() ? 'Existed' : 'Success';
    }

    return response()->json($response);
}

    public function invoice($id)
    {
        $this->pageTitle = __('app.menu.invoices');
        $this->pageIcon = 'icon-money';

        $this->invoice = Invoice::with('currency', 'project', 'project.client')->whereRaw('md5(id) = ?', $id)->firstOrFail();
        $this->paidAmount = $this->invoice->getPaidAmount();

        if ($this->invoice->discount > 0) {
            if ($this->invoice->discount_type == 'percent') {
                $this->discount = (($this->invoice->discount / 100) * $this->invoice->sub_total);
            } else {
                $this->discount = $this->invoice->discount;
            }
        } else {
            $this->discount = 0;
        }

        $taxList = array();

        $items = InvoiceItems::whereNotNull('taxes')
            ->where('invoice_id', $this->invoice->id)
            ->get();

        foreach ($items as $item) {
            foreach (json_decode($item->taxes) as $tax) {
                $this->tax = InvoiceItems::taxbyid($tax)->first();
                if ($this->tax) {
                    if (!isset($taxList[$this->tax->tax_name . ': ' . $this->tax->rate_percent . '%'])) {
                        $taxList[$this->tax->tax_name . ': ' . $this->tax->rate_percent . '%'] = ($this->tax->rate_percent / 100) * $item->amount;
                    } else {
                        $taxList[$this->tax->tax_name . ': ' . $this->tax->rate_percent . '%'] = $taxList[$this->tax->tax_name . ': ' . $this->tax->rate_percent . '%'] + (($this->tax->rate_percent / 100) * $item->amount);
                    }
                }
            }
        }

        $this->taxes = $taxList;

        $this->settings = Setting::organisationSetting();
        $this->credentials = PaymentGatewayCredentials::first();
        $this->methods = OfflinePaymentMethod::activeMethod();
        $this->invoiceSetting = InvoiceSetting::first();

        return view('invoice', [
            'companyName' => $this->settings->company_name,
            'pageTitle' => $this->pageTitle,
            'pageIcon' => $this->pageIcon,
            'global' => $this->settings,
            'setting' => $this->settings,
            'settings' => $this->settings,
            'invoice' => $this->invoice,
            'paidAmount' => $this->paidAmount,
            'discount' => $this->discount,
            'credentials' => $this->credentials,
            'taxes' => $this->taxes,
            'methods' => $this->methods,
            'invoiceSetting' => $this->invoiceSetting,
        ]);
    }

    public function ganttData($ganttProjectId)
    {

        $data = array();
        $links = array();

        $projects = Project::select('id', 'project_name', 'start_date', 'deadline', 'completion_percent')
            ->whereRaw('md5(id) = ?', $ganttProjectId)
            ->get();

        $id = 0; //count for gantt ids
        foreach ($projects as $project) {
            $id = $id + 1;
            $projectId = $id;

            // TODO::ProjectDeadline to do
            $projectDuration = 0;
            if ($project->deadline) {
                $projectDuration = $project->deadline->diffInDays($project->start_date);
            }

            $data[] = [
                'id' => $projectId,
                'text' => ucwords($project->project_name),
                'start_date' => $project->start_date->format('Y-m-d H:i:s'),
                'duration' => $projectDuration,
                'progress' => $project->completion_percent / 100,
                'color' => 'grey',
                'textColor' => 'white',
                'project_id' => $project->id
            ];

            $tasks = Task::projectTasks($project->id, null, "0");

            foreach ($tasks as $key => $task) {
                $id = $id + 1;

                $taskDuration = $task->due_date->diffInDays($task->start_date);
                $taskDuration = $taskDuration + 1;

                $color = $task->board_column->label_color;

                $data[] = [
                    'id' => $task->id,
                    'text' => ucfirst($task->heading),
                    'start_date' => (!is_null($task->start_date)) ? $task->start_date->format('Y-m-d') : $task->due_date->format('Y-m-d'),
                    'duration' => $taskDuration,
                    'parent' => $projectId,
                    'color' => $color,
                    'taskid' => $task->id
                ];

                $links[] = [
                    'id' => $id,
                    'source' => $task->dependent_task_id != '' ? $task->dependent_task_id : $projectId,
                    'target' => $task->id,
                    'type' => $task->dependent_task_id != '' ? 0 : 1
                ];
            }
        }

        $ganttData = [
            'data' => $data,
            'links' => $links
        ];

        return response()->json($ganttData);
    }

// Function to get the name of the person who voted based on the vote ID
function getNameByCount($data, $vote) {
    // If $data is a JSON string, decode it
    if (is_string($data)) {
        $data = json_decode($data, true); // Convert JSON string to array
    }

    // Check if $data is an array and has a 'messages' key
    if (is_array($data) && isset($data['messages'])) {
        // Iterate through the messages in the data
        foreach ($data['messages'] as $message) {
            // Check if the message contains a poll with results
            if (isset($message['poll']['results'])) {
                // Search for the matching vote ID in the poll results
                foreach ($message['poll']['results'] as $result) {
                    if ($result['id'] === $vote) {
                        return $result['name']; // Return the name if a match is found
                    }
                }
            }
        }
    }

    // Return null if no match is found
    return null;
}

private function getOrCreateCronGroup($groupName)
{
    $apiKey = "30082a82048b865e3cc3b745b4ecd864";

    // Step 1: Check if group exists
    $ch = curl_init("https://api.easycron.com/v1/cron-job-groups");
    curl_setopt_array($ch, [
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_HTTPHEADER => [
            "Content-Type: application/json",
            "X-API-Key: $apiKey"
        ]
    ]);

    $response = curl_exec($ch);
    curl_close($ch);

    $data = json_decode($response, true);

    if (!empty($data['cron_job_groups'])) {
        foreach ($data['cron_job_groups'] as $group) {
            if ($group['cron_job_group_name'] === $groupName) {
                // ✅ Group exists
                return $group['cron_job_group_id'];
            }
        }
    }

    // Step 2: Create group if not found
    $payload = json_encode(['cron_job_group_name' => $groupName]);

    $ch = curl_init("https://api.easycron.com/v1/cron-job-groups");
    curl_setopt_array($ch, [
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_POST => true,
        CURLOPT_HTTPHEADER => [
            "Content-Type: application/json",
            "X-API-Key: $apiKey"
        ],
        CURLOPT_POSTFIELDS => $payload
    ]);

    $response = curl_exec($ch);
    $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);

    $data = json_decode($response, true);

    if ($httpCode === 201 && isset($data['cron_job_group_id'])) {
        return $data['cron_job_group_id'];
    }

    \Log::warning("❌ Failed to create or fetch group: $groupName");
    return null;
}

public function deletefu()
{
    $apiKey = "30082a82048b865e3cc3b745b4ecd864";
    $threshold = now()->subHours(7);

    $records = AITrackerFollowup::where('created_at', '<', $threshold)->get();

    foreach ($records as $record) {
        $jobId = $record->id_job;

        if (!$jobId) continue;

        $deleteUrl = "https://api.easycron.com/v1/cron-jobs/{$jobId}";

        $ch = curl_init();
        curl_setopt_array($ch, [
            CURLOPT_URL => $deleteUrl,
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_CUSTOMREQUEST => 'DELETE',
            CURLOPT_HTTPHEADER => [
                "Content-Type: application/json",
                "X-API-Key: $apiKey"
            ]
        ]);

        $response = curl_exec($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        curl_close($ch);

        if ($httpCode === 204) { // 204 = success delete
            $record->delete();
            \Log::info("✅ Deleted EasyCron Job: $jobId & Tracker Row");
        } else {
            \Log::warning("❌ Failed to delete EasyCron Job: $jobId | HTTP: $httpCode | Response: $response");
        }
    }

    return 'success';
}

 public function failed(\Exception $exception = null)
{
    // Log the failure
    Log::error('ProcessChatGPT job failed after retries.', [
        'id_staff' => $this->id_staff,
        'idz' => $this->idz,
        'wano' => $this->wano ?? 'N/A', // Check if $wano is set
        'exception' => $exception ? $exception->getMessage() : null,
    ]);
    
    $user = Zuser::where('id_staff', '=', $this->id_staff)->first();

    // Send a fallback message to the customer
    if (!empty($this->wano)) {
        $this->sendMessage(
            $this->wano,
            'Kurang jelas dengan soalan cik, boleh jelaskan lagi?',
            $this->instance,
            1
        );
    } else {
        Log::warning('Failed job cannot send message because wa_no is not set.');
    }
}

    $final_timestamp = (int)$masa - 3600 + $randomSeconds;
    $current =  (int)$masa + $randomSeconds;
